export type ScheduledExercise = {
  key: string; // unique within the program: w{week}-d{day}-{exerciseId}
  exercise: Exercise;
  role: Role;
  sets: number;
  reps: number;

export type ScheduledDay = {
  week: number; // 1..4
  day: number; // 1..7 inside the week
  number: number; // 1..28 across the program
  rest: boolean;
  title: string;
  focus: string;
  exercises: ScheduledExercise[];
  coreMinutes: number;
  optionalMinutes: number;
  totalMinutes: number;
};
 
export function buildSchedule(program: Program): ScheduledDay[] {
  const byId = new Map(program.exercises.map((e) => [e.id, e]));
  const days: ScheduledDay[] = [];
  for (let week = 1; week <= WEEKS; week++) {
    const scheme = program.weekSchemes[week - 1];
    for (let day = 1; day <= DAYS_PER_WEEK; day++) {
      const number = (week - 1) * DAYS_PER_WEEK + day;
      if (day > WORKOUT_DAYS_PER_WEEK) {
        days.push({
          week,
          day,
          number,
          rest: true,
          title: "Rest day",
          focus: "Recover and stretch lightly",
          exercises: [],
          coreMinutes: 0,
          optionalMinutes: 0,
          totalMinutes: 0,
        });
        continue;
      }
      const tpl = program.days[day - 1];
      const exercises: ScheduledExercise[] = tpl.items.map(
        ({ exerciseId, role }) => {
          const exercise = byId.get(exerciseId);
          Iif (!exercise)
            throw new Error(
              `Unknown exercise ${exerciseId} in ${program.slug}`,
            );
          return {
            key: `w${week}-d${day}-${exerciseId}`,
            exercise,
            role,
            sets: scheme.sets,
            reps: scheme.reps,
            minutes: Math.round(exercise.baseMinutes * scheme.minutesFactor),
          };
        },
      );
      const core = exercises
        .filter((e) => e.role === "core")
        .reduce((s, e) => s + e.minutes, 0);
      const opt = exercises
        .filter((e) => e.role === "accessory")
        .reduce((s, e) => s + e.minutes, 0);
      days.push({
        week,
        day,
        number,
        rest: false,
        title: tpl.title,
        focus: tpl.focus,
        exercises,
        coreMinutes: core,
        optionalMinutes: opt,
        totalMinutes: core + opt,
      });
    }
  }
  return days;
}
 
export function minutesRange(days: ScheduledDay[]): [number, number] {
  const m = days.filter((d) => !d.rest).map((d) => d.totalMinutes);
  return [Math.min(...m), Math.max(...m)];
}
